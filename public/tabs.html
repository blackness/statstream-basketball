Supabase setup SQL

Run this in Database → SQL Editor → New query:

create table if not exists tabs (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  artist text not null,
  genre text,
  difficulty integer default 3,
  key text,
  tuning text default 'Standard',
  tags text[],
  source text,
  tab text,
  notes text,
  created_at timestamptz default now()
);
alter table tabs enable row level security;
create policy "Public read"   on tabs for select using (true);
create policy "Public insert" on tabs for insert with check (true);
create policy "Public delete" on tabs for delete using (true);
create policy "Public update" on tabs for update using (true);
